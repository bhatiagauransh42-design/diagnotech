### COMPONENT 1 — ScrambleLinkButton

Use this component as the authoritative source for the CTA/link microinteraction. Preserve its behavior and API; adapt only styling/context needed to fit the existing product.

```tsx
"use client"

import * as React from "react"
import { ArrowRight } from "lucide-react"

export interface ScrambleLinkButtonProps
  extends React.AnchorHTMLAttributes<HTMLAnchorElement> {
  btnText?: string
  href?: string
  textClassName?: string
  hoverColor?: string
  showLine?: boolean
  lineClassName?: string
  showArrow?: boolean
  icon?: React.ReactNode
  iconClassName?: string
  scrambleDuration?: number
  stepMs?: number
  revealStagger?: number
  onClick?: React.MouseEventHandler<HTMLAnchorElement>
}

const GLYPHS = "abcdefghijklmnopqrstuvwxyz0123456789"

export function ScrambleLinkButton({
  btnText = "Hover me",
  href = "#",
  className,
  textClassName,
  hoverColor = "#ff6b00",
  showLine = true,
  lineClassName,
  showArrow = true,
  icon,
  iconClassName,
  scrambleDuration = 500,
  stepMs = 30,
  revealStagger = 1,
  onClick,
  ...linkProps
}: ScrambleLinkButtonProps) {
  const [text, setText] = React.useState(btnText)
  const [hovered, setHovered] = React.useState(false)
  const reduced = React.useRef(false)

  React.useEffect(() => {
    setText(btnText)
  }, [btnText])

  React.useEffect(() => {
    if (typeof window !== "undefined") {
      reduced.current = window.matchMedia("(prefers-reduced-motion: reduce)").matches
    }
  }, [])

  const scramble = React.useCallback(() => {
    if (reduced.current) {
      setText(btnText)
      return
    }

    const target = btnText
    const steps = Math.max(1, Math.ceil(scrambleDuration / stepMs))
    let step = 0

    const id = window.setInterval(() => {
      step += 1
      const revealed = Math.min(target.length, step * revealStagger)
      const next = target
        .split("")
        .map((char, i) => {
          if (i < revealed || char === " ") return char
          return GLYPHS[Math.floor(Math.random() * GLYPHS.length)]
        })
        .join("")

      setText(next)

      if (step >= steps) {
        window.clearInterval(id)
        setText(target)
      }
    }, stepMs)

    return () => window.clearInterval(id)
  }, [btnText, scrambleDuration, stepMs, revealStagger])

  React.useEffect(() => {
    if (hovered) return scramble()
  }, [hovered, scramble])

  return (
    <a
      href={href}
      {...linkProps}
      onClick={onClick}
      onMouseEnter={() => setHovered(true)}
      onMouseLeave={() => setHovered(false)}
      onFocus={() => setHovered(true)}
      onBlur={() => setHovered(false)}
      className={[
        "group inline-flex items-center gap-2 text-sm uppercase tracking-[0.14em]",
        "transition-colors duration-300",
        className,
      ].filter(Boolean).join(" ")}
      style={
        {
          "--scramble-hover": hoverColor,
        } as React.CSSProperties
      }
    >
      {icon ? <span className={iconClassName}>{icon}</span> : null}
      <span className={textClassName}>{text}</span>
      {showLine ? (
        <span
          aria-hidden
          className={[
            "h-px w-8 bg-current transition-all duration-300 group-hover:w-12",
            lineClassName,
          ].filter(Boolean).join(" ")}
        />
      ) : null}
      {showArrow ? (
        icon ?? <ArrowRight className="size-4 -rotate-45 transition-transform duration-300 group-hover:rotate-0" />
      ) : null}
    </a>
  )
}

export default ScrambleLinkButton
```

Required package: `lucide-react`.

Important: if the exact source supplied by the user exists in the project/attachments, prefer that exact source over this reconstruction. Do not invent a different interaction.
